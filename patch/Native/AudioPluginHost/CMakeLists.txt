cmake_minimum_required(VERSION 3.22)
project(PatchAudioPluginHost LANGUAGES C CXX OBJC OBJCXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_OSX_DEPLOYMENT_TARGET 14.0)
add_subdirectory("${JUCE_SOURCE}" juce EXCLUDE_FROM_ALL)
add_library(PatchAudioPluginHost SHARED PluginHost.cpp LoopmotherBridge.cpp Loopmother/SynthEngine.cpp)
target_compile_definitions(PatchAudioPluginHost PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_PLUGINHOST_VST=0 JUCE_PLUGINHOST_AU=0 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_MODAL_LOOPS_PERMITTED=1)
target_link_libraries(PatchAudioPluginHost PRIVATE juce::juce_audio_processors juce::juce_recommended_config_flags)
set_target_properties(PatchAudioPluginHost PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_NAME_DIR "@rpath")
