project(VJNative LANGUAGES CXX)

  cmake_policy(SET CMP0167 NEW)

set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(nlohmann_json REQUIRED)

find_package(Threads REQUIRED)
add_library(vj_application src/model.cpp src/library_placement.cpp src/meta_mapping.cpp src/api_server.cpp src/remote_server.cpp src/mapping_canvas.cpp src/render_scene.cpp src/mapping_model.cpp src/mapping_geometry.cpp src/mapping_collections.cpp src/media_library.cpp src/control_model.cpp src/live_sources.cpp src/ui_commands.cpp src/text_source.cpp src/collection_rendering.cpp src/output_model.cpp src/trace_model.cpp src/midi_runtime.cpp src/composition_api.cpp src/fixture_output.cpp src/module_registry.cpp src/room_model.cpp src/frame_transport.cpp src/calibration_model.cpp src/calibration_detector.cpp src/calibration_scan.cpp src/media_processing.cpp src/api_compat.cpp src/mapping_compat.cpp src/ndi_clients.cpp src/removable_media.cpp src/calibration_debug.cpp src/media_http.cpp)

  find_package(ALSA QUIET)
endif()

  target_sources(vj_application PRIVATE src/midi_alsa.cpp)
  target_link_libraries(vj_application PRIVATE ALSA::ALSA)
else()
  target_sources(vj_application PRIVATE src/midi_stub.cpp)

option(VJ_PORTABLE_TEXT "Build FreeType/HarfBuzz raster and outline text backends" OFF)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND (VJ_BUILD_ENGINE OR VJ_BUILD_MAC_UI))
  set(VJ_PORTABLE_TEXT ON)
endif()

  find_package(Freetype REQUIRED)
  find_package(PNG REQUIRED)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(HARFBUZZ REQUIRED IMPORTED_TARGET harfbuzz)
  target_sources(vj_application PRIVATE src/freetype_text.cpp)
  target_link_libraries(vj_application PUBLIC Freetype::Freetype PNG::PNG PkgConfig::HARFBUZZ)
  target_compile_definitions(vj_application PUBLIC VJ_HAS_PORTABLE_TEXT=1)

include(cmake/Link.cmake)
vj_configure_link(vj_application)

target_link_libraries(vj_application PUBLIC nlohmann_json::nlohmann_json Boost::headers Threads::Threads ${CMAKE_DL_LIBS})
target_compile_definitions(vj_application PRIVATE BOOST_ERROR_CODE_HEADER_ONLY VJ_BUNDLED_FONT_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/../static/fonts" VJ_API_ASSET_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/..")
target_compile_options(vj_application PRIVATE -Wall -Wextra -Wpedantic)
add_executable(vj-headless src/headless.cpp)
target_link_libraries(vj-headless PRIVATE vj_application)
install(TARGETS vj-headless RUNTIME DESTINATION bin)
set(VJ_API_ASSETS docs/external-api.md vector-effects/sdk/abi.ts vector-effects/sdk/graphics-v2.ts vector-effects/sdk/math.ts vector-effects/effects/external-example.ts static/vector/runtime.js)
foreach(asset IN LISTS VJ_API_ASSETS)
  get_filename_component(asset_dir "${asset}" DIRECTORY)
  install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../${asset}" DESTINATION "share/vj-native/${asset_dir}")
endforeach()
include(CTest)

  target_link_libraries(vj-mapping-test PRIVATE vj_application)

  add_executable(vj-mapping-canvas-test tests/mapping_canvas_test.cpp)

if(VJ_BUILD_ENGINE OR VJ_BUILD_MAC_UI)
  find_package(PkgConfig REQUIRED)
  find_package(OpenGL REQUIRED)
  find_package(JPEG REQUIRED)
  pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
  pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil libswscale libswresample libavdevice)
  pkg_check_modules(AVFILTER QUIET IMPORTED_TARGET libavfilter)
  set(RENDERER ${CMAKE_CURRENT_SOURCE_DIR}/../native-renderer/src)
  set(VJ_WASMTIME_ROOT "" CACHE PATH "Wasmtime distribution root")
  if(NOT VJ_WASMTIME_ROOT)
    include(${CMAKE_CURRENT_SOURCE_DIR}/../native-renderer/cmake/Wasmtime.cmake)
    set(VJ_WASMTIME_HINTS "${vibe_wasmtime_SOURCE_DIR}")
  endif()
  find_path(VJ_WASMTIME_INCLUDE_DIR wasmtime.h HINTS ${VJ_WASMTIME_ROOT} ${VJ_WASMTIME_HINTS} PATH_SUFFIXES include REQUIRED)
  find_library(VJ_WASMTIME_LIBRARY wasmtime HINTS ${VJ_WASMTIME_ROOT} ${VJ_WASMTIME_HINTS} PATH_SUFFIXES lib REQUIRED)
  add_library(vj_preview_engine src/render_demand.cpp src/calibration_debug_codec.cpp src/frame_publisher.cpp src/frame_codec.cpp src/room_render_pass.cpp src/calibration_capture.cpp src/vector_host.cpp src/mapping_render_pass.cpp src/trace_render_geometry.cpp src/led_sampler.cpp src/mapped_outputs.cpp ${RENDERER}/ndi_output.cpp ${RENDERER}/native_render_graph.cpp ${RENDERER}/media_decoder.cpp ${RENDERER}/live_capture.cpp ${RENDERER}/native_audio.cpp ${RENDERER}/ndi_input.cpp ${RENDERER}/ndi_runtime.cpp)
  target_include_directories(vj_preview_engine PUBLIC src PRIVATE ${VJ_WASMTIME_INCLUDE_DIR})
  target_include_directories(vj_preview_engine PUBLIC ${RENDERER} ${RENDERER}/../third_party/ndi/include)
  target_link_libraries(vj_preview_engine PUBLIC vj_application ${VJ_WASMTIME_LIBRARY} OpenGL::GL PkgConfig::SDL2 PkgConfig::FFMPEG JPEG::JPEG nlohmann_json::nlohmann_json Threads::Threads ${CMAKE_DL_LIBS})
  if(AVFILTER_FOUND)
    target_compile_definitions(vj_preview_engine PRIVATE VIBE_HAS_AVFILTER=1)
    target_link_libraries(vj_preview_engine PRIVATE PkgConfig::AVFILTER)
  endif()
  include(${RENDERER}/../cmake/MappedMetal.cmake)
  if(APPLE)
    target_link_libraries(vj_preview_engine PUBLIC vj_mapped_metal)
  endif()
  include(cmake/LiveSources.cmake)
  vj_add_live_source_backend()
  if(APPLE)
    add_library(vj_midi_backend src/midi_mac.mm)
    target_link_libraries(vj_midi_backend PUBLIC vj_application "-framework CoreMIDI" "-framework Foundation")
    set_target_properties(vj_midi_backend PROPERTIES OBJCXX_STANDARD 20)
    target_link_libraries(vj-headless PRIVATE vj_midi_backend)
    target_compile_definitions(vj-headless PRIVATE VJ_HAS_MAC_MIDI=1)
    add_library(vj_text_backend src/mac_text_source.mm)
    target_link_libraries(vj_text_backend PUBLIC vj_application "-framework CoreFoundation" "-framework CoreText" "-framework CoreGraphics" "-framework ImageIO")
    set_target_properties(vj_text_backend PROPERTIES OBJCXX_STANDARD 20)
  endif()
  if(BUILD_TESTING)
    add_executable(vj-mapping-render-test tests/mapping_render_test.cpp)
    target_link_libraries(vj-mapping-render-test PRIVATE vj_preview_engine)
    add_test(NAME native_mapping_render COMMAND vj-mapping-render-test)
    add_executable(vj-composer-source tests/composer_source.cpp)
    target_link_libraries(vj-composer-source PRIVATE vj_preview_engine)
  endif()
  add_library(vj_render_host src/render_host.cpp src/media_metadata.cpp)
  target_link_libraries(vj_render_host PUBLIC vj_application vj_preview_engine)
  add_executable(vj-render-service src/render_service.cpp)
  target_link_libraries(vj-render-service PRIVATE vj_render_host)
  if(APPLE)
    target_link_libraries(vj-render-service PRIVATE vj_text_backend vj_midi_backend)
    target_compile_definitions(vj-render-service PRIVATE VJ_HAS_MAC_MIDI=1)
  endif()
  install(DIRECTORY ../static/fonts/ DESTINATION share/vj-native/fonts)
  install(TARGETS vj-render-service RUNTIME DESTINATION bin)
endif()
if(VJ_BUILD_MAC_UI)
  if(NOT APPLE)
    message(FATAL_ERROR "VJ_BUILD_MAC_UI requires macOS; use vj-headless on Linux")
  endif()
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 20)
  add_executable(VJNative MACOSX_BUNDLE src/main.mm src/native_async.mm src/native_style.mm src/native_forms.mm src/native_color.mm src/native_performance.mm src/native_widgets.mm src/native_library.mm src/native_clip_search.mm src/app.mm src/preview.mm src/mapper.mm src/native_meta_mapping.mm src/native_media_browser.mm src/native_controls.mm src/native_source_browser.mm src/native_outputs.mm src/media_metadata.cpp src/native_midi.mm src/native_room.mm src/native_calibration.mm src/native_thumbnails.mm src/native_network.mm src/show_library.mm)
  target_compile_options(VJNative PRIVATE -fobjc-arc -Wall -Wextra -Wno-deprecated-declarations)
  target_link_libraries(VJNative PRIVATE vj_application vj_preview_engine vj_text_backend vj_midi_backend "-framework AppKit" "-framework OpenGL" "-framework CoreMIDI" "-framework Foundation" "-framework CoreImage")
  add_custom_command(TARGET VJNative POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_SOURCE_DIR}/../static/fonts $<TARGET_BUNDLE_CONTENT_DIR:VJNative>/Resources/fonts)
  add_custom_command(TARGET VJNative POST_BUILD COMMAND ${CMAKE_COMMAND} -E rm -f $<TARGET_BUNDLE_CONTENT_DIR:VJNative>/Resources/vector/modules/electric-strips.wasm $<TARGET_BUNDLE_CONTENT_DIR:VJNative>/Resources/vector/modules/laser-text.wasm $<TARGET_BUNDLE_CONTENT_DIR:VJNative>/Resources/vector/modules/lorenz-attractor.wasm $<TARGET_BUNDLE_CONTENT_DIR:VJNative>/Resources/vector/modules/slime-mold.wasm $<TARGET_BUNDLE_CONTENT_DIR:VJNative>/Resources/vector/modules/vector-cube.wasm $<TARGET_BUNDLE_CONTENT_DIR:VJNative>/Resources/vector/modules/vector-scan-lines.wasm)
  add_custom_command(TARGET VJNative POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different ${CMAKE_CURRENT_SOURCE_DIR}/VJ.icns $<TARGET_BUNDLE_CONTENT_DIR:VJNative>/Resources/VJ.icns)
  foreach(asset IN LISTS VJ_API_ASSETS)
    get_filename_component(asset_dir "${asset}" DIRECTORY)
    add_custom_command(TARGET VJNative POST_BUILD COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_BUNDLE_CONTENT_DIR:VJNative>/Resources/${asset_dir}" COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/../${asset}" "$<TARGET_BUNDLE_CONTENT_DIR:VJNative>/Resources/${asset}")
  endforeach()
  add_dependencies(VJNative vj-source-discovery)
  add_custom_command(TARGET VJNative POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:vj-source-discovery> $<TARGET_FILE_DIR:VJNative>/vj-source-discovery)
  include(cmake/Patch.cmake)
  # Seal resources after copying them: an executable-only linker signature is
  # not a valid app signature, and macOS rejects its Bonjour trust record.
  add_custom_command(TARGET VJNative POST_BUILD COMMAND /usr/bin/codesign --force --sign - "$<TARGET_BUNDLE_DIR:VJNative>" VERBATIM)
  set_target_properties(VJNative PROPERTIES OUTPUT_NAME VJ MACOSX_BUNDLE_INFO_PLIST ${CMAKE_CURRENT_SOURCE_DIR}/Info.plist)
endif()
